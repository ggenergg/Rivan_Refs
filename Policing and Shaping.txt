### MARKING

| PC  | QOS  |
| --- | ---  |
| NOC | AF41 |
| SOC | AF33 |
| HR  | CS2  |


~~~
!@R1
conf t
 ip access-list extended ACL-NOC
  permit ip host 192.168.103.101 any
 ip access-list extended ACL-SOC
  permit ip host 192.168.103.102 any
 ip access-list extended ACL-HR
  permit ip host 192.168.103.103 any
 !
 class-map match-any CM-NOC
  match access-group name ACL-NOC
 class-map match-any CM-SOC
  match access-group name ACL-SOC
 class-map match-any CM-HR
  match access-group name ACL-HR
 !
 policy-map PM1
  class CM-NOC
   set dscp AF41 
  class CM-SOC
   set dscp AF33
  class CM-HR
   set dscp CS2
  class class-default
   set dscp default
 int g3
  no service-policy input PM1
  service-policy input PM1
  end
show policy-map int g3
~~~

### CONGESTION AVOIDANCE

~~~
!@R1-EDGE
conf t
 class-map match-all VIP
  match dscp AF41 
 class-map match-all NORMAL
  match dscp AF33
 class-map match-all SCAVENGE
  match dscp CS2
 !
 NO policy-map PM5
 policy-map PM5
  class VIP
   priority 640               
  class NORMAL
   shape average 400000
   !
   bandwidth 160              
   random-detect dscp-based  
   random-detect ecn         
  class SCAVENGE
   shape average 400000
   !
   bandwidth 80            
   random-detect dscp-based
   random-detect ecn
   !
   random-detect dscp CS2 
  class class-default
   !
 int g1
  service-policy output PM5
  end
~~~

# Control Plane Policing


### Prevent ICMP Overload

~~~
!@R1
config t
 no ip access-list extended PING
 ip access-list extended PING
  permit icmp any any
 !
 class-map CM-PING
  match access-group name PING
 !
 no policy-map PM-PING
 policy-map PM-PING
  class CM-PING
   police 8000 
    conform-action transmit 
	exceed-action drop
 !
 control-plane
  service-policy input PM-PING
  end
sh policy-map control-plane
~~~


Verify:
~~~
!@cmd
ping 208.8.8.11 -t -l 65500
ping 208.8.8.11 -t -l 1000
ping 208.8.8.11 -t -l 1200
~~~


<br>


~~~
!@R1
conf t
 control-plane
  no service-policy input PM-PING
  end
~~~
