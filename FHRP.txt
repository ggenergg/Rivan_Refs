### PRECONFIGS

~~~
!@UTM-PH  R1
conf t
 hostname UTM-PH--R1
 enable secret pass
 service password-encryption
 no logging cons
 ip domain lookup
 ip domain lookup source-interface G2
 ip name-server 192.168.102.8
 line vty 0 14
  transport input all
  password pass
  login local
  exec-timeout 0 0
 int g1
  ip add 208.8.8.11 255.255.255.0
  ip nat outside
  no shut
 int g2
  ip add 192.168.102.11 255.255.255.0
  ip nat inside
  no shut
 int g3
  ip add 192.168.103.11 255.255.255.0
  ip nat inside
  no shut
 !
 ip access-list extended NAT
  permit ip 192.168.102.0 0.0.0.255 any
  permit ip 192.168.103.0 0.0.0.255 any
 !
 ip nat inside source list NAT int g1 overload
 !
 username admin privilege 15 secret pass
 ip http server
 ip http secure-server
 ip http authentication local
 ip route 0.0.0.0 0.0.0.0 208.8.8.2
 end
wr
!
~~~

## HSRP

~~~
!@R1 (PRIMARY INTERNET)
conf t
 track 1 int gi 1 line-protocol
 !
 int gi 3
  standby 10 ip 192.168.103.69
  standby 10 preempt
  standby 10 priority 120
  standby 10 track 1 decrement 60
  end
show standby
~~~

~~~
!@R2 (BACKUP)
conf t
 track 1 int gi 1 line-protocol
 !
 int gi 3
  standby 10 ip 192.168.103.69
  standby 10 preempt
  standby 10 priority 110
  standby 10 track 1 decrement 60
  end
show standby
~~~

~~~
!@R3 (BACKUP of BACKUP / Last Resort)
conf t
 int gi 3
  standby 10 ip 192.168.103.69
  standby 10 priority 100
  end
show standby
~~~

### HSRPv1 vs HSRPv2

 !

### GLBP

~~~
!@R1 (PRIMARY INTERNET)  AVG
conf t
 track 3 int gi 1 line-protocol
 !
 int gi 3
  no vrrp 10
  glbp 10 ip 192.168.103.69
  glbp 10 preempt
  glbp 10 priority 120
  glbp 10 load-balancing round-robin
  glbp 10 weighting 120 lower 70 upper 100
  glbp 10 weighting track 3 decrement 60
  end
